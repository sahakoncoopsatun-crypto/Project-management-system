การเชื่อมต่อฐานข้อมูล Google Sheets & Apps Script

ระบบรองรับการรับส่งข้อมูลกับ **Google Sheets** และ **Google Forms** โดยอัตโนมัติ

โค้ด Google Apps Script (Code.gs) สำหรับคัดลอกไปวาง:

function doGet(e) {
  var action = e.parameter.action || 'getProjects';
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (action === 'getProjects') {
    var sheet = ss.getSheetByName('Projects');
    var data = sheet ? sheet.getDataRange().getValues() : [];
    return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var projSheet = ss.getSheetByName('Projects') || ss.insertSheet('Projects');
    projSheet.appendRow([new Date(), data.id, data.title, data.fiscalYear, data.purpose, JSON.stringify(data.budgetItems || []), JSON.stringify(data.participants || [])]);
    return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
