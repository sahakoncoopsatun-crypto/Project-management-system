การเชื่อมต่อฐานข้อมูล Google Sheets & Apps Script

ระบบรองรับการรับส่งข้อมูลกับ **Google Sheets** และ **Google Forms** โดยอัตโนมัติ ไม่ต้องเสียค่าใช้จ่ายเซิร์ฟเวอร์

โค้ด Google Apps Script (Code.gs) สำหรับคัดลอกไปวาง:

// Google Apps Script (Code.gs)
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet();
  var action = e.parameter.action || 'getProjects';
  
  if (action === 'getProjects') {
    var data = sheet.getSheetByName('Projects').getDataRange().getValues();
    return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Projects');
  sheet.appendRow([new Date(), data.id, data.title, data.budget, data.fiscalYear]);
  return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
}
